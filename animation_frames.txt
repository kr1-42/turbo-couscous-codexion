static void	ft_putstr(const char *s)
{
	write(1, s, ft_strlen(s));
}

static void	*input_listener(void *arg)
{
	int		*flag;
	char	c;

	flag = (int *)arg;
	while (read(0, &c, 1) > 0)
	{
		if (c == '1')
		{
			*flag = 1;
			break ;
		}
	}
	return (NULL);
}

static void	render_frame(int f)
{
	int	i;

	ft_putstr("\033[H");
	i = 0;
	while (i < FRAME_H)
		ft_putstr(g_frames[f][i++]);
}

int	exit_screen(void)
{
	int			f;
	int			*flag;
	int			dir;
	pthread_t	thread;

	flag = malloc(sizeof(int));
	*flag = 0;
	pthread_create(&thread, NULL, input_listener, flag);
	ft_putstr("\033[2J");
	f = 0;
	dir = 1;
	while (1)
	{
		render_frame(f);
		if (*flag)
			break ;
		usleep(DELAY_US);
		f += dir;
		if (f >= 11)
			dir = -1;
		if (f <= 0)
			dir = 1;
	}
	pthread_join(thread, NULL);
	return (free(flag), 0);
}
